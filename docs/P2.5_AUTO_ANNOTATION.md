# P2.5 任务：自动化图纸缺陷标注（Cursor 全自动版）

## 🎯 任务目标

让 Cursor **自动审查** `data/drawings/real_samples/` 的 18 份 DXF 图纸，标注出违反给排水设计规范的缺陷，建立 `drawing_p1` 评测集。

## ⚠️ 自动标注的限制说明

由于 LLM 无法可视化图纸，只能分析文本化的实体列表，因此：
- 可能**漏检**某些需要空间判断的缺陷
- 位置描述可能不如人工准确
- 对于复杂的系统性问题可能判断不准

**解决方案**：为每个缺陷标记**信心度**（高/中/低），人工只需快速复核低信心案例。

#!/usr/bin/env python3
"""自动审查图纸并标注缺陷。"""
import json
import sys
from pathlib import Path
from typing import List, Dict, Any

# 添加项目路径
sys.path.insert(0, str(Path(__file__).parent.parent))

from water_agent.drawing.dxf_reader import read_dxf_file


def load_drawing(dxf_path: Path) -> Dict[str, Any]:
    """读取 DXF 图纸。"""
    result = read_dxf_file(str(dxf_path))
    if not result["ok"]:
        raise ValueError(f"无法读取图纸: {result.get('error')}")
    return result


def check_pipe_diameter(entities: List[Dict]) -> List[Dict]:
    """检查管径是否符合规范。"""
    defects = []
    
    # 提取所有文本实体
    texts = [e for e in entities if e["type"] == "TEXT"]
    
    for text in texts:
        content = text.get("text", "").strip()
        
        # 匹配管径标注（DN + 数字）
        if content.startswith("DN") and content[2:].isdigit():
            diameter = int(content[2:])
            layer = text.get("layer", "")
            
            # 规则 1：生活给水立管 < DN20
            if "给水" in layer and diameter < 20:
                defects.append({
                    "category": "管径不符",
                    "severity": "高",
                    "description": f"生活给水管 {content} 不符合规范要求",
                    "location": {
                        "layer": layer,
                        "entity_type": "TEXT",
                        "entity_content": content,
                        "approx_position": f"图层 {layer}",
                    },
                    "regulation": {
                        "spec": "GB50015-2019",
                        "clause": "3.5.2",
                        "requirement": "生活给水立管管径不应小于 DN20"
                    },
                    "expected": "DN20 或更大",
                    "actual": content,
                    "confidence": "中",  # 中等信心：无法确认是立管还是支管
                    "notes": "自动检测：假设为立管，请人工验证"
                })
            
            # 规则 2：排水立管 < DN75
            if "排水" in layer and diameter < 75:
                defects.append({
                    "category": "管径不符",
                    "severity": "高",
                    "description": f"排水管 {content} 不符合规范要求",
                    "location": {
                        "layer": layer,
                        "entity_type": "TEXT",
                        "entity_content": content,
                        "approx_position": f"图层 {layer}",
                    },
                    "regulation": {
                        "spec": "GB50015-2019",
                        "clause": "4.3.2",
                        "requirement": "住宅排水立管管径不应小于 DN75"
                    },
                    "expected": "DN75 或更大",
                    "actual": content,
                    "confidence": "中",
                    "notes": "自动检测：假设为住宅排水立管，请人工验证"
                })
            
            # 规则 3：消防管 < DN100
            if "消防" in layer and diameter < 100:
                defects.append({
                    "category": "管径不符",
                    "severity": "高",
                    "description": f"消防管 {content} 不符合规范要求",
                    "location": {
                        "layer": layer,
                        "entity_type": "TEXT",
                        "entity_content": content,
                        "approx_position": f"图层 {layer}",
                    },
                    "regulation": {
                        "spec": "GB50015-2019",
                        "clause": "8.2.2",
                        "requirement": "室内消火栓管径不应小于 DN100"
                    },
                    "expected": "DN100 或更大",
                    "actual": content,
                    "confidence": "高",  # 高信心：消防管规范明确
                    "notes": "自动检测：消防管径不足"
                })
    
    return defects


def check_material(entities: List[Dict]) -> List[Dict]:
    """检查管道材料是否符合规范。"""
    defects = []
    
    texts = [e for e in entities if e["type"] == "TEXT"]
    
    for text in texts:
        content = text.get("text", "").strip().upper()
        layer = text.get("layer", "")
        
        # 规则 1：热水管使用 PVC
        if "热水" in layer and "PVC" in content:
            defects.append({
                "category": "材料选用错误",
                "severity": "高",
                "description": "热水管使用 PVC 不符合规范要求",
                "location": {
                    "layer": layer,
                    "entity_type": "TEXT",
                    "entity_content": content,
                    "approx_position": f"图层 {layer}",
                },
                "regulation": {
                    "spec": "GB50015-2019",
                    "clause": "3.5.4",
                    "requirement": "热水管道应采用耐热材料，不得使用 PVC-U 管"
                },
                "expected": "PP-R、铜管或不锈钢管",
                "actual": "PVC",
                "confidence": "高",
                "notes": "自动检测：热水管材料不符"
            })
        
        # 规则 2：消防管使用塑料管
        if "消防" in layer and any(plastic in content for plastic in ["PPR", "PVC", "PE"]):
            defects.append({
                "category": "材料选用错误",
                "severity": "高",
                "description": "消防管使用塑料管不符合规范要求",
                "location": {
                    "layer": layer,
                    "entity_type": "TEXT",
                    "entity_content": content,
                    "approx_position": f"图层 {layer}",
                },
                "regulation": {
                    "spec": "GB50015-2019",
                    "clause": "8.2.3",
                    "requirement": "消防管道应采用钢管或不锈钢管"
                },
                "expected": "镀锌钢管或不锈钢管",
                "actual": content,
                "confidence": "高",
                "notes": "自动检测：消防管材料不符"
            })
    
    return defects


def check_slope(entities: List[Dict]) -> List[Dict]:
    """检查排水管坡度是否符合规范。"""
    defects = []
    
    texts = [e for e in entities if e["type"] == "TEXT"]
    
    for text in texts:
        content = text.get("text", "").strip()
        layer = text.get("layer", "")
        
        # 匹配坡度标注（i=0.XXX 或 i=1:XXX）
        if content.startswith("i=") or content.startswith("I="):
            slope_str = content[2:].strip()
            
            # 解析坡度
            slope = None
            if ":" in slope_str:
                # i=1:100 格式
                parts = slope_str.split(":")
                if len(parts) == 2 and parts[1].isdigit():
                    slope = 1.0 / float(parts[1])
            elif slope_str.replace(".", "").isdigit():
                # i=0.012 格式
                slope = float(slope_str)
            
            if slope is None:
                continue
            
            # 尝试从附近找到管径
            # 简化处理：假设排水管
            if "排水" in layer:
                # 规则：DN100 排水管坡度 < 0.012
                if slope < 0.012:
                    defects.append({
                        "category": "坡度不足",
                        "severity": "中",
                        "description": f"排水管坡度 {content} 不符合规范要求",
                        "location": {
                            "layer": layer,
                            "entity_type": "TEXT",
                            "entity_content": content,
                            "approx_position": f"图层 {layer}",
                        },
                        "regulation": {
                            "spec": "GB50015-2019",
                            "clause": "4.3.5",
                            "requirement": "DN100 排水横管坡度不应小于 0.012"
                        },
                        "expected": "i ≥ 0.012",
                        "actual": content,
                        "confidence": "低",  # 低信心：未确认管径
                        "notes": "自动检测：假设为 DN100，未能匹配具体管径，请人工验证"
                    })
    
    return defects


def annotate_drawing(dxf_path: Path, output_path: Path):
    """自动标注单张图纸。"""
    
    print(f"正在审查: {dxf_path.name}")
    
    # 读取图纸
    drawing = load_drawing(dxf_path)
    entities = drawing.get("entities", [])
    
    # 判断图纸类型
    drawing_id = dxf_path.stem
    if drawing_id.startswith("P-"):
        drawing_type = "平面图"
    elif drawing_id.startswith("S-"):
        drawing_type = "系统图"
    elif drawing_id.startswith("D-"):
        drawing_type = "大样图"
    elif drawing_id.startswith("PUMP-"):
        drawing_type = "泵房图"
    else:
        drawing_type = "未知"
    
    # 运行检查规则
    all_defects = []
    all_defects.extend(check_pipe_diameter(entities))
    all_defects.extend(check_material(entities))
    all_defects.extend(check_slope(entities))
    
    # 去重（基于描述）
    seen = set()
    unique_defects = []
    for defect in all_defects:
        key = (defect["category"], defect["description"], defect["location"]["entity_content"])
        if key not in seen:
            seen.add(key)
            unique_defects.append(defect)
    
    # 分配缺陷 ID
    for i, defect in enumerate(unique_defects, start=1):
        defect["defect_id"] = f"{drawing_id}-D{i:03d}"
    
    # 生成标注文件
    annotation = {
        "drawing_id": drawing_id,
        "drawing_type": drawing_type,
        "annotation_date": "2026-09-15",
        "annotator": "自动审查（Cursor）",
        "total_defects": len(unique_defects),
        "defects": unique_defects,
        "notes": "此文件由 LLM 自动生成，建议人工复核信心度为'低'或'中'的案例"
    }
    
    # 保存
    output_path.parent.mkdir(parents=True, exist_ok=True)
    with open(output_path, "w", encoding="utf-8") as f:
        json.dump(annotation, f, ensure_ascii=False, indent=2)
    
    print(f"  ✓ 发现 {len(unique_defects)} 个缺陷")
    
    # 统计信心度
    confidence_counts = {"高": 0, "中": 0, "低": 0}
    for defect in unique_defects:
        confidence_counts[defect.get("confidence", "低")] += 1
    
    print(f"    信心度分布: 高={confidence_counts['高']}, 中={confidence_counts['中']}, 低={confidence_counts['低']}")


def main():
    """批量标注所有图纸。"""
    
    drawings_dir = Path("data/drawings/real_samples")
    annotations_dir = Path("data/eval/drawing_p1/annotations")
    
    # 获取所有 DXF 文件
    dxf_files = sorted(drawings_dir.glob("*.dxf"))
    
    print(f"找到 {len(dxf_files)} 份图纸\n")
    
    for dxf_file in dxf_files:
        output_file = annotations_dir / f"{dxf_file.stem}.json"
        try:
            annotate_drawing(dxf_file, output_file)
        except Exception as e:
            print(f"  ✗ 标注失败: {e}")
    
    print(f"\n✅ 自动标注完成！")
    print(f"   输出目录: {annotations_dir}")
    print(f"\n⚠️  建议人工复核信心度为'低'或'中'的案例")


if __name__ == "__main__":
    main()

## 🎯 自动标注的规则

脚本会自动检测以下缺陷：

### 1. 管径不符
- 给水管 < DN20 → 高/中信心（无法区分立管/支管）
- 排水管 < DN75 → 中信心（假设为立管）
- 消防管 < DN100 → 高信心

### 2. 材料错误
- 热水管使用 PVC → 高信心
- 消防管使用塑料管 → 高信心

### 3. 坡度不足
- 排水管坡度 < 0.012 → 低信心（未匹配具体管径）

**限制**：
- ❌ 标高检查：需要空间坐标计算，暂不支持
- ❌ 缺少设施：需要图纸全局分析，暂不支持
- ❌ 间距检查：需要计算实体间距离，暂不支持
