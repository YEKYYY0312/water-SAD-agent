🔒 CSRF Token 测试页面

测试给排水 Agent Web API 的 CSRF 保护机制

⚠️ 使用前请确保：
  1. 服务已启动（docker compose up -d）
  2. 已通过 login.html 登录

1️⃣ 检查 CSRF Token

2️⃣ 测试 GET 请求（不需要 CSRF Token）

3️⃣ 测试 POST 请求（需要 CSRF Token）

4️⃣ 代码示例

正确的 API 调用方式：

// 1. 获取 CSRF token
function getCsrfToken() {
    const match = document.cookie.match(/wa_csrf=([^;]+)/);
    return match ? match[1] : null;
}

// 2. 发送需要 CSRF 保护的请求
async function createProject(name) {
    const csrfToken = getCsrfToken();
    const response = await fetch('http://127.0.0.1:8000/api/projects', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'X-CSRF-Token': csrfToken  // 关键：必须带上这个 header
        },
        credentials: 'include',
        body: JSON.stringify({ name, description: 'Test' })
    });
    return response.json();
}
